\begin{thebibliography}{99}

\bibitem[*]{rsi-forge}
RSI-Forge: a multi-agent pipeline (construction, reproduction, review) that
converts published papers into executable environments for self-improvement.
210 environments across 18 fields; 90 reviewed by human domain experts.
Preprint.

\bibitem{harbor}
Laude Institute. \emph{Harbor}: a framework for evaluating and optimizing
agents and language models in container environments.
\url{https://pypi.org/project/harbor/}

\bibitem[*]{terminalbench}
Terminal-Bench 2.0: a benchmark for agentic terminal use, with an emphasis on
task validation quality.

\bibitem[*]{drugdiscoverybench}
Scale AI. DrugDiscoveryBench: 82 verifiable, domain-expert-curated tasks in
early-stage drug discovery, formatted for Harbor.

\bibitem[*]{tbscience}
Terminal-Bench-Science: 70 expert-curated research workflow tasks spanning
life sciences, physics, and earth sciences.

\bibitem[*]{bioagentgym}
BioAgent Gym: a biomedical agent repository with a native Harbor leaderboard.

\bibitem[*]{rsiexam}
RSI-Exam: evaluation infrastructure running on Harbor for long-horizon
self-improvement and generalization to unseen data.

\bibitem[*]{compilebench}
CompileBench: agent evaluation for compiling open-source projects, migrated to
Harbor.

\bibitem[*]{bixbench}
BixBench: 50+ real-world scenarios and nearly 300 open-ended questions in
computational biology.

\bibitem[*]{scienceagentbench}
ScienceAgentBench: 102 scientific data-analysis tasks across computational
biology, geoscience, psychology, and materials science.

\bibitem[*]{labbench}
LAB-Bench: a benchmark of literature-scale biological reasoning tasks.

\bibitem[*]{rlvr}
Reinforcement learning with verifiable rewards (RLVR) for equipping LLMs with
agentic capabilities.

\bibitem[*]{gvpo}
Group verification-based policy optimization (GVPO).

\bibitem[*]{entropyaware}
Entropy-aware reinforcement learning with verifiable rewards.

\bibitem[*]{rar}
Rubrics as Rewards (RaR): structured, checklist-style rubrics as interpretable
reward signals where no single standard answer exists.

\bibitem[*]{contamination}
On benchmark contamination and the inflation of performance estimates from
test data present in training corpora.

\bibitem[*]{harnessoptbench}
HarnessOpt-Bench: measuring whether models can self-modify their execution
harness.

\bibitem[*]{officeqa}
OfficeQA: a benchmark for question answering over office documents.

\bibitem[*]{gaia}
GAIA: a benchmark for general AI assistants.

\bibitem[*]{claroaibench}
ClaroAI-Bench: a benchmark study of biomedical paper reproduction, finding
environment reconstruction to be the dimension with highest evaluator
disagreement.

\bibitem{europepmc}
Europe PMC: programmatic access to over 37~million publications, including
3.2~million fully open-access articles.
\url{https://europepmc.org/RestfulWebService}

\end{thebibliography}
